# 1. Baixar o código
git clone <https://github.com/franciscopignateli-sketch/Domus-Fit.git>

# 2. Entrar na pasta
cd nome-da-pasta

# 3. Instalar as dependências (OBRIGATÓRIO)
npm install

# 4. Preparar xampp
Abrir a pasta C:\xampp\htdocs
Copiar a pasta domus_backend (que veio do git)
Colar na pasta C:\xampp\htdocs

# 5. Preparar php my admin
importar base dados domus-fit.sql no php my admin

# 6. Iniciar o site
npm run dev
